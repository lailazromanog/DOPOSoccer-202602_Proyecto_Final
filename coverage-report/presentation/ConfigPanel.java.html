package presentation;

import domain.DOPOSoccerException;
import domain.Game;
import javax.swing.JFileChooser;
import java.awt.Color;
import java.awt.Component;
import java.awt.Dimension;
import java.awt.Font;
import java.awt.Graphics;
import java.awt.GridBagLayout;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.IOException;
import javax.imageio.ImageIO;
import javax.swing.BorderFactory;
import javax.swing.Box;
import javax.swing.BoxLayout;
import javax.swing.ButtonGroup;
import javax.swing.JComboBox;
import javax.swing.JLabel;
import javax.swing.JOptionPane;
import javax.swing.JPanel;
import javax.swing.JRadioButton;
import javax.swing.JTextField;

/**
 * Pantalla de configuracion del partido
 * Deja elegir el tipo de partido, por tiempo o por goles y el valor.
 */

public class ConfigPanel extends JPanel {

    private BufferedImage background;
    private JRadioButton byTimeOption;
    private JRadioButton byGoalsOption;
    private JLabel valueLabel;
    private JTextField valueField;
    private RoundedButton playButton;
    private RoundedButton openButton;
    private RoundedButton shopButton;
    private JComboBox<Resolution> resolutionBox;
    private JComboBox<String> opponentBox;


    /**
     * Crea la pantalla de configuración del partido, con su fondo, sus
     * componentes y las acciones de sus botones ya listas.
     * @param gui ventana principal que recibe la orden de cambiar de pantalla
     */
    public ConfigPanel(GameGUI gui){
        loadBackground();
        prepareElements();
        prepareActions(gui);
    }

    /**
     * Carga la imagen de fondo de la pantalla de configuración desde
     * el archivo de recursos del juego.
     */
    private void loadBackground(){
        try{
            background = ImageIO.read(new File("resources/menu.png"));
        }catch(IOException e){
            System.out.println("No se pudo cargar el fondo: " + e.getMessage());
        }
    }
    /**
     * Dibuja la imagen de fondo de la pantalla, si se pudo cargar.
     * @param g contexto gráfico sobre el que se dibuja el panel
     */
    @Override
    protected void paintComponent(Graphics g){
        super.paintComponent(g);
        if(background != null){
            g.drawImage(background, 0, 0, getWidth(), getHeight(), this);
        }
    }

    /**
     * Crea y ubica todos los componentes de la pantalla de
     * configuración dentro de la tarjeta central: tipo de partido,
     * valor del partido, resolución, rival y los botones de jugar,
     * abrir partida y mejorar estadísticas.
     */
    private void prepareElements(){
        setPreferredSize(new Dimension(769, 600));
        setLayout(new GridBagLayout()); // centra la tarjeta en la pantalla

        // tarjeta oscura para que todo se lea sobre la foto
        JPanel card = new JPanel();
        card.setBackground(new Color(0x14212a));
        card.setLayout(new BoxLayout(card, BoxLayout.Y_AXIS));
        card.setBorder(BorderFactory.createCompoundBorder(
                BorderFactory.createLineBorder(new Color(0xF28C28), 3, true),
                BorderFactory.createEmptyBorder(26, 48, 26, 48)));

        JLabel subtitle = label("Configuracion del partido", new Font("Arial", Font.BOLD, 26), Color.WHITE);

        //tipo de partido
        byTimeOption = radio("Por tiempo", true);
        byGoalsOption = radio("Por goles", false);
        ButtonGroup group = new ButtonGroup();
        group.add(byTimeOption);
        group.add(byGoalsOption);

        //tiempo de la partida o numero de goles que elija el usuario
        valueLabel = label("Minutos: ", new Font("Arial", Font.PLAIN, 16), new Color(0xbcd4df));
        valueField = new JTextField("3", 5);
        valueField.setMaximumSize(new Dimension(90, 34));
        valueField.setFont(new Font("Arial", Font.BOLD, 16));
        valueField.setAlignmentX(Component.CENTER_ALIGNMENT);
        
        resolutionBox = new JComboBox<>(Resolution.values());
        resolutionBox.setSelectedItem(Resolution.MEDIUM); // mediana por defecto
        resolutionBox.setMaximumSize(new Dimension(220, 32));
        resolutionBox.setAlignmentX(Component.CENTER_ALIGNMENT);
        
        String[] opponents = {"Humano", "Torpe (facil)", "Ofensiva (media)", "Defensiva (media)", "Experta (dificil)"};
        opponentBox = new JComboBox<>(opponents);
        opponentBox.setSelectedIndex(0); // humano por defecto
        opponentBox.setMaximumSize(new Dimension(220, 32));
        opponentBox.setAlignmentX(Component.CENTER_ALIGNMENT);

        playButton = new RoundedButton("JUGAR", new Color(0xF28C28));
        playButton.setFont(new Font("Arial", Font.BOLD, 22));
        playButton.setMaximumSize(new Dimension(220, 52));
        playButton.setPreferredSize(new Dimension(220, 52));
        playButton.setAlignmentX(Component.CENTER_ALIGNMENT);


        openButton = new RoundedButton("Abrir partida", new Color(0x3a4a57));
        openButton.setFont(new Font("Arial", Font.BOLD, 16));
        openButton.setMaximumSize(new Dimension(220, 40));
        openButton.setPreferredSize(new Dimension(220, 40));
        openButton.setAlignmentX(Component.CENTER_ALIGNMENT);

        shopButton = new RoundedButton("Mejorar estadisticas", new Color(0x3a4a57));
        shopButton.setFont(new Font("Arial", Font.BOLD, 16));
        shopButton.setMaximumSize(new Dimension(220, 40));
        shopButton.setPreferredSize(new Dimension(220, 40));
        shopButton.setAlignmentX(Component.CENTER_ALIGNMENT);

        // todo va en la card
        card.add(subtitle);
        card.add(Box.createVerticalStrut(22));
        card.add(byTimeOption);
        card.add(Box.createVerticalStrut(4));
        card.add(byGoalsOption);
        card.add(Box.createVerticalStrut(20));
        card.add(valueLabel);
        card.add(Box.createVerticalStrut(6));
        card.add(valueField);
        card.add(Box.createVerticalStrut(16));
        card.add(label("Resolucion:", new Font("Arial", Font.PLAIN, 14), new Color(0xbcd4df)));
        card.add(resolutionBox);
        card.add(Box.createVerticalStrut(12));
        card.add(label("Rival:", new Font("Arial", Font.PLAIN, 14), new Color(0xbcd4df)));
        card.add(opponentBox);
        card.add(Box.createVerticalStrut(22));
        card.add(playButton);
        card.add(Box.createVerticalStrut(10));
        card.add(openButton);
        card.add(Box.createVerticalStrut(8));
        card.add(shopButton);

        add(card);
    }

    /**
     * crea cada boton en la pantalla
     */
    /**
     * Asigna a cada botón y cada opción de la pantalla la acción que
     * debe ejecutar cuando el jugador la selecciona.
     * @param gui ventana principal que recibe la orden de cambiar de pantalla
     */
    private void prepareActions(final GameGUI gui){
        byTimeOption.addActionListener(new ActionListener(){
            /**
             * Cambia la etiqueta del valor a mostrar minutos, para la
             * opción de partido por tiempo.
             * @param e evento de selección de la opción por tiempo
             */
            @Override
            public void actionPerformed(ActionEvent e){
                valueLabel.setText("Minutos: ");
            }
        });

        byGoalsOption.addActionListener(new ActionListener(){
            /**
             * Cambia la etiqueta del valor a mostrar goles para ganar,
             * para la opción de partido por goles.
             * @param e evento de selección de la opción por goles
             */
            @Override
            public void actionPerformed(ActionEvent e){
                valueLabel.setText("Goles para ganar: ");
            }
        });


        openButton.addActionListener(new ActionListener(){
            /**
             * Abre el selector de archivos y, si el jugador elige uno,
             * intenta cargar la partida guardada en él y entrar a la
             * cancha con esa partida y la resolución elegida. Si el
             * archivo no contiene una partida válida, muestra el error
             * al jugador y lo registra en el log.
             * @param e evento de clic del botón abrir partida
             */
            @Override
            public void actionPerformed(ActionEvent e){
                JFileChooser chooser = new JFileChooser();
                int result = chooser.showOpenDialog(ConfigPanel.this);
                if(result == JFileChooser.APPROVE_OPTION){
                    try {
                        Game game = Game.openFrom(chooser.getSelectedFile());
                        Resolution res = (Resolution) resolutionBox.getSelectedItem();
                        gui.openGame(game, res.getWidth(), res.getHeight());
                    } catch (DOPOSoccerException ex) {
                        GameLogger.error("No se pudo abrir la partida", ex);
                        JOptionPane.showMessageDialog(ConfigPanel.this, ex.getMessage(), "Error", JOptionPane.ERROR_MESSAGE);
                    }
                }
            }
        });

        shopButton.addActionListener(new ActionListener(){
            /**
             * Muestra la tienda de mejoras de estadísticas.
             * @param e evento de clic del botón mejorar estadísticas
             */
            @Override
            public void actionPerformed(ActionEvent e){
                gui.showShop();
            }
        });


        playButton.addActionListener(new ActionListener(){
            /**
             * Reúne el tipo de partido, el valor escrito, la
             * resolución y el rival elegidos, y pasa a la pantalla de
             * selección de personajes con esos datos. Si el valor
             * escrito no es válido, muestra el error al jugador y lo
             * registra en el log sin cambiar de pantalla.
             * @param e evento de clic del botón jugar
             */
            @Override
            public void actionPerformed(ActionEvent e){
                try {
                    String matchType;
                    if (byTimeOption.isSelected()) {
                        matchType = "time";
                    } else {
                        matchType = "goals";
                    }
                    int value = readValue();          // puede lanzar DOPOSoccerException
                    Resolution res = (Resolution) resolutionBox.getSelectedItem();
                    String opponent;
                    int oi = opponentBox.getSelectedIndex();
                    if (oi == 0) {
                        opponent = "human";
                    }else if (oi == 1) {
                        opponent = "stupid";
                    }else if (oi == 2) {
                        opponent = "offensive";
                    }else if(oi == 3){
                        opponent = "defensive";
                    }else{
                        opponent = "expert";
                    }
                    gui.showCharacterSelect(matchType, value, opponent, res.getWidth(), res.getHeight());
                }catch (DOPOSoccerException ex) {
                    GameLogger.error("Configuracion invalida del partido", ex);  // para los programadores
                    JOptionPane.showMessageDialog(ConfigPanel.this,ex.getMessage(), "Error", JOptionPane.ERROR_MESSAGE); // para el usuario
                }
            }
        });
    }

    /**
     * Lee el número escrito en el campo de texto del valor del
     * partido y lo valida.
     * @return el valor escrito, ya validado
     * @throws DOPOSoccerException si el texto no es un número entero válido, o si no es mayor que cero
     */
    private int readValue() throws DOPOSoccerException{
        String text = valueField.getText().trim();
        try{
            int v = Integer.parseInt(text);
            if(v <= 0){
                throw new DOPOSoccerException("El valor debe ser un numero mayor que 0. Escribiste: ");
            }
            return v;
        }catch(NumberFormatException ex){
            throw new DOPOSoccerException("El valor debe ser un numero entero valido.");
        }
    }

    //pa no repetir codigo
    /**
     * Crea una etiqueta de texto con la fuente y el color dados, para
     * no repetir la misma configuración en cada etiqueta de la pantalla.
     * @param text texto de la etiqueta
     * @param font fuente de la etiqueta
     * @param color color del texto de la etiqueta
     * @return la etiqueta creada
     */
    private JLabel label(String text, Font font, Color color){
        JLabel l = new JLabel(text);
        l.setFont(font);
        l.setForeground(color);
        l.setAlignmentX(Component.CENTER_ALIGNMENT);
        return l;
    }

    /**
     * Crea un botón de selección única con el texto dado, para no
     * repetir la misma configuración en cada opción de la pantalla.
     * @param text texto de la opción
     * @param selected verdadero para que la opción empiece seleccionada
     * @return el botón de selección creado
     */
    private JRadioButton radio(String text, boolean selected){
        JRadioButton r = new JRadioButton(text, selected);
        r.setFont(new Font("Arial", Font.PLAIN, 16));
        r.setForeground(Color.WHITE);
        r.setOpaque(false);
        r.setAlignmentX(Component.CENTER_ALIGNMENT);
        return r;
    }
}
